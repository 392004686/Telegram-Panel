## 第二步：复制下面的整个代码块，粘贴到SSH终端执行

```bash
cd /opt/Telegram-Panel && \
echo "=========================================="  && \
echo "步骤 1: 更新代码" && \
echo "=========================================="  && \
git fetch origin && \
git checkout codex/multi-user-ui && \
git pull origin codex/multi-user-ui && \
echo "当前commit:" && \
git log -1 --oneline && \
echo "" && \
echo "==========================================" && \
echo "步骤 2: 备份数据" && \
echo "==========================================" && \
TIMESTAMP=$(date +%Y%m%d-%H%M%S) && \
BACKUP_FILE="/root/telegram-panel-data-${TIMESTAMP}.tar.gz" && \
echo "创建备份: $BACKUP_FILE" && \
tar -czf "$BACKUP_FILE" docker-data/ && \
ls -lh "$BACKUP_FILE" && \
sha256sum "$BACKUP_FILE" && \
echo "" && \
echo "==========================================" && \
echo "步骤 3: 构建Docker镜像" && \
echo "==========================================" && \
COMMIT_HASH=$(git rev-parse --short HEAD) && \
IMAGE_TAG="telegram-panel:multi-user-ui-${COMMIT_HASH}" && \
echo "构建镜像: $IMAGE_TAG" && \
docker build -t "$IMAGE_TAG" -t "telegram-panel:latest" . && \
echo "" && \
echo "==========================================" && \
echo "步骤 4: 停止旧容器（只停5000端口）" && \
echo "==========================================" && \
docker compose stop telegram-panel && \
docker compose rm -f telegram-panel && \
echo "" && \
echo "==========================================" && \
echo "步骤 5: 更新配置并启动新容器" && \
echo "==========================================" && \
sed -i 's|^TP_IMAGE=.*|TP_IMAGE=telegram-panel:latest|' .env 2>/dev/null || echo 'TP_IMAGE=telegram-panel:latest' > .env && \
grep -q '^TP_UPDATE_MODE=' .env && sed -i 's/^TP_UPDATE_MODE=.*/TP_UPDATE_MODE=image/' .env || echo 'TP_UPDATE_MODE=image' >> .env && \
echo "启动容器..." && \
docker compose up -d telegram-panel && \
echo "等待容器启动..." && \
sleep 10 && \
echo "" && \
echo "==========================================" && \
echo "步骤 6: 验证部署" && \
echo "==========================================" && \
echo "容器状态:" && \
docker compose ps telegram-panel && \
echo "" && \
echo "使用的镜像:" && \
docker inspect telegram-panel --format '{{.Config.Image}}' && \
echo "" && \
echo "健康检查:" && \
curl -sf http://127.0.0.1:5000/healthz && echo "✓ 5000端口健康检查通过" || echo "✗ 5000端口健康检查失败" && \
echo "" && \
echo "验证7000端口未受影响:" && \
curl -sf http://127.0.0.1:7000/healthz && echo "✓ 7000端口正常" || echo "7000端口无响应（如果没有运行7000是正常的）" && \
echo "" && \
echo "容器日志（最后30行）:" && \
docker logs --tail 30 telegram-panel && \
echo "" && \
echo "==========================================" && \
echo "✅ 部署完成！" && \
echo "==========================================" && \
echo "5000端口: http://16.216.65.42:5000" && \
echo "备份文件: $BACKUP_FILE" && \
echo "镜像标签: $IMAGE_TAG" && \
echo "=========================================="
```
